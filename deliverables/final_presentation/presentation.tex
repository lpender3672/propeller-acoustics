\begin{frame}{Acoustic scaling}
    \begin{itemize}
        \item<1-> How is noise of a given propeller quantised?
        \[
            p_\text{rms} =  \sqrt{ \frac{1}{T} \int_0^T p(t)^2\, dt}
        \]
        \item<2-> How does this change with size $(r)$, speed $(\Omega)$ and observer position $(R,\theta)$?
        \[
        p_\text{rms} = 
        \only<1>{\sqrt{ \frac{1}{T} \int_0^T p(t)^2\, dt }
                \phantom{ = \frac{\rho(r \Omega)^2}{R/r} \left| g(\theta, \text{geometry}, ?) \right|}}
        \only<2>{\frac{\rho(r \Omega)^2}{\phantom{R/r}}
                \phantom{\left| g(\theta, \text{geometry}, ?) \right|}}
        \only<3>{\frac{\rho(r \Omega)^2}{R/r}
                \phantom{\left| g(\theta, \text{geometry}, ?) \right|}}
        \only<4->{\frac{\rho(r \Omega)^2}{R/r}
                \left| g(\theta, \text{geometry}, ?) \right|}
        \]
    \end{itemize}
\end{frame}


\begin{frame}{Aerodynamic scaling}
    \begin{itemize}
        
        \item How does a given propellers aerodynamic performance change?
        \[
            T = C_T \cdot \rho \Omega^2 r^4
        \]
        \[
            \only<1->{P = C_P \cdot \rho \Omega^3 r^5}
        \]
        \item<2> Propulsive efficiency in forward flight
        \[
            \only<2>{\eta = \frac{2 V}{V + V_e}}
        \]
        \vspace{-2.3cm}
        \item<3> Propulsive efficiency at hover
        \[
            \only<3->{FM = \frac{\text{Ideal hover power}}{\text{Actual hover power}} = \frac{C_T^{3/2}}{\sqrt{2}C_Q}}
        \]
        
    \end{itemize}
\end{frame}

\begin{frame}{Comparison at reference performance}
    Given 
    \begin{itemize}
        \item test propeller with coefficients $C_T$ and $C_Q$,
        \item reference propeller $(C_T', C_Q')$,
    \end{itemize}
    \uncover<1->{
    An effective radius $(r_{ef})$ and speed $(\Omega_{ef})$ for the test propeller can be found for same $(T, P)$.
    \[
    \frac{r_{ef}}{r} = \frac{FM}{FM'} \quad \quad \quad \frac{\Omega_{ef}}{\Omega} = \left(\frac{C_T}{C_T'}\right)^{5/2} \left( \frac{C_Q'}{C_Q} \right)^2
    \]}
    \uncover<2->{
    These then scale the test propeller noise the reference propeller.
    \[    
    p_\text{rms}' = p_\text{rms} \left(\frac{r_{ef}}{r}\right)^3 \left( \frac{\Omega_{ef}}{\Omega}\right)^2 = p_\text{rms} \left(\frac{C_T'}{C_T}\right)^{7/2} \left(\frac{C_Q}{C_Q'} \right)^4
    \]}
    
\end{frame}

\begin{frame}{Hanson's Theory \cite{hanson_theory}}
        
    \begin{equation}
        \begin{aligned}
            g_m = \frac{1}{4\pi} \exp\left(jmB \left( \frac{\Omega R}{c_0} - \frac{\pi}{2} \right)\right) \\
            \times \int_0^1 z^2 \exp \left(j z \frac{r_t}{b} \left(k_y \sin \gamma + k_x \cos \gamma\right) \right) \\
            \times J_{mB} \left( \frac{mB z \Omega r_t}{c_0} \sin \theta \right)
                jk_y (C_L / 2) \Psi_L(k_x) dz
        \end{aligned}
        \label{eq:hansons_result}
    \end{equation}

    The terms $k_x$ and $k_y$ represent non-dimensional wavenumbers in the $x$ and $y$ directions respectively.

    \begin{equation}
        k_x = \frac{mB b}{z r_t}, \quad
        k_y = \frac{mB\Omega b}{z c_0} \cos \theta, \quad
    \end{equation}

    \begin{equation}
        \Psi_L(k_x) = \int_{-1/2}^{1/2} L(X) \exp(jk_x X) dX
    \end{equation}
\end{frame}
