\newcommand{\styledEllipse}[5]{%
  % left half: from top (90°) down to bottom (270°)
  \draw[#4]
    ($(#1)+(0,#3)$)
    arc[
      x radius=#2,
      y radius=#3,
      start angle=90,
      end angle=270
    ];%
  % right half: from bottom (–90°) up to top (90°)
  \draw[#5]
    ($(#1)+(0,-#3)$)
    arc[
      x radius=#2,
      y radius=#3,
      start angle=-90,
      end angle=90
    ];%
}
